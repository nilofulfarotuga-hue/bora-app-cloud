## §1. Migration M1 — Reversão `bora_role` + adaptação do trigger

### Propósito
- Reverter `raw_user_meta_data.bora_role` do admin de `'admin'` → `'client'` (fix da regressão dos 3 logins).
- Substituir trigger `trg_protect_admin_bora_role` da Fase 1 por outro que **proteja `raw_app_meta_data.role='admin'`** (defesa em camadas — apesar de `raw_app_meta_data` já ser write-only via service_role).
- Confirmar `raw_app_meta_data.role='admin'` continua intacto.

### SQL exacto

```sql
-- Migration: 20260428000004_admin_role_to_app_metadata.sql
BEGIN;

-- 1.1 Revert bora_role for the admin user — back to 'client' so the user
-- can log in via loginClientAsync again. The admin gate moves to
-- raw_app_meta_data.role which is already populated.
UPDATE auth.users
SET raw_user_meta_data =
  COALESCE(raw_user_meta_data, '{}'::jsonb)
  || '{"bora_role":"client"}'::jsonb
WHERE email = 'nilofulfarotuga@gmail.com';

-- 1.2 Belt-and-braces: ensure raw_app_meta_data.role IS 'admin' for the
-- admin user. If for any reason it was cleared, re-set it. Do NOT touch
-- existing keys (provider, providers, …).
UPDATE auth.users
SET raw_app_meta_data =
  COALESCE(raw_app_meta_data, '{}'::jsonb)
  || '{"role":"admin"}'::jsonb
WHERE email = 'nilofulfarotuga@gmail.com';

-- 1.3 Replace the Phase-1 trigger to protect raw_app_meta_data.role.
-- raw_app_meta_data is normally only writable by the service_role, but
-- we keep this as defense-in-depth in case anyone runs a migration that
-- merges metadata wholesale.
DROP TRIGGER IF EXISTS trg_protect_admin_bora_role ON auth.users;
DROP FUNCTION IF EXISTS public.protect_admin_bora_role();

CREATE OR REPLACE FUNCTION public.protect_admin_app_role()
RETURNS TRIGGER
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public, auth
AS $$
DECLARE
  v_old_role TEXT;
  v_new_role TEXT;
  v_email    TEXT;
BEGIN
  v_email := COALESCE(OLD.email, NEW.email);
  -- Lock-list of protected admin emails
  IF v_email <> 'nilofulfarotuga@gmail.com' THEN
    RETURN NEW;
  END IF;

  v_old_role := OLD.raw_app_meta_data ->> 'role';
  v_new_role := NEW.raw_app_meta_data ->> 'role';

  -- If admin role was set and someone tries to demote/clear it, restore.
  IF v_old_role = 'admin' AND COALESCE(v_new_role, '') <> 'admin' THEN
    NEW.raw_app_meta_data :=
      COALESCE(NEW.raw_app_meta_data, '{}'::jsonb)
      || '{"role":"admin"}'::jsonb;
  END IF;

  RETURN NEW;
END;
$$;

DROP TRIGGER IF EXISTS trg_protect_admin_app_role ON auth.users;
CREATE TRIGGER trg_protect_admin_app_role
  BEFORE UPDATE ON auth.users
  FOR EACH ROW
  EXECUTE FUNCTION public.protect_admin_app_role();

COMMENT ON FUNCTION public.protect_admin_app_role() IS
  'Phase-2-B replacement of protect_admin_bora_role: protects raw_app_meta_data.role=admin instead of user_metadata.bora_role. raw_app_meta_data is write-only by service_role; this trigger is defense-in-depth.';

COMMIT;
```

## §2. Migration M2 — Gates server-side passam para `app_metadata.role`

### Lista de objectos a migrar

| # | Objecto | Mudança |
|---|---|---|
| 2.1 | RPC `_admin_op_guard()` | trocar `user_metadata.bora_role` → `app_metadata.role` |
| 2.2 | RLS `public.admin_audit_log` policy `admin_audit_log_select_admin` | mesmo padrão |
| 2.3 | RLS `storage.objects` policy `admin_read_all_driver_docs` | usa `users.raw_user_meta_data` (DB join) → trocar para `users.raw_app_meta_data->>'role'` |
| 2.4 | RPC `admin_dashboard_metrics()` | adicionar gate via `_admin_op_guard()` (era unguarded — gap encontrado em B.1.4) |

### SQL exacto

```sql
-- Migration: 20260428000005_admin_gate_app_metadata.sql
BEGIN;

-- 2.1 Re-define _admin_op_guard to read from app_metadata.role
CREATE OR REPLACE FUNCTION public._admin_op_guard()
RETURNS TABLE (admin_id UUID, admin_email TEXT)
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_uid   UUID;
  v_email TEXT;
  v_role  TEXT;
BEGIN
  v_uid := auth.uid();
  IF v_uid IS NULL THEN
    RAISE EXCEPTION 'admin_required: not authenticated' USING ERRCODE = '42501';
  END IF;

  v_email := COALESCE(
    auth.jwt() ->> 'email',
    auth.jwt() -> 'user_metadata' ->> 'email'
  );

  -- Primary: app_metadata.role (immutable by client)
  -- Fallback: user_metadata.bora_role (Phase-1 legacy, kept for one
  --   version so old sessions still work after the migration)
  v_role := COALESCE(
    auth.jwt() -> 'app_metadata'  ->> 'role',
    auth.jwt() -> 'user_metadata' ->> 'bora_role',
    auth.jwt() ->> 'role'
  );

  IF v_role IS DISTINCT FROM 'admin' THEN
    RAISE EXCEPTION 'admin_required: caller role=% (need admin)',
      COALESCE(v_role, 'NULL') USING ERRCODE = '42501';
  END IF;

  RETURN QUERY SELECT v_uid, v_email;
END;
$$;

REVOKE ALL ON FUNCTION public._admin_op_guard() FROM public, anon;
GRANT EXECUTE ON FUNCTION public._admin_op_guard() TO authenticated;

-- 2.2 Update RLS policy on admin_audit_log
DROP POLICY IF EXISTS "admin_audit_log_select_admin" ON public.admin_audit_log;
CREATE POLICY "admin_audit_log_select_admin"
  ON public.admin_audit_log
  FOR SELECT
  TO authenticated
  USING (
    COALESCE(
      auth.jwt() -> 'app_metadata'  ->> 'role',
      auth.jwt() -> 'user_metadata' ->> 'bora_role',
      auth.jwt() ->> 'role'
    ) = 'admin'
  );

-- 2.3 Update storage.objects policy admin_read_all_driver_docs
-- (uses DB join because storage.objects RLS evaluates server-side
-- against auth.users directly, not the JWT claims).
DROP POLICY IF EXISTS "admin_read_all_driver_docs" ON storage.objects;
CREATE POLICY "admin_read_all_driver_docs"
  ON storage.objects
  FOR SELECT
  TO authenticated
  USING (
    bucket_id = 'driver-documents'
    AND EXISTS (
      SELECT 1 FROM auth.users
      WHERE users.id = auth.uid()
        AND (
          users.raw_app_meta_data  ->> 'role'      = 'admin'  -- canonical
          OR users.raw_user_meta_data ->> 'bora_role' = 'admin'  -- legacy
        )
    )
  );

-- 2.4 Add gate to admin_dashboard_metrics (was unguarded)
CREATE OR REPLACE FUNCTION public.admin_dashboard_metrics()
RETURNS JSON
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_admin RECORD;
  v_result JSON;
BEGIN
  SELECT admin_id, admin_email INTO v_admin FROM public._admin_op_guard();
  -- (the previous body — read-only aggregates)
  SELECT json_build_object(
    'platform_revenue', (
      SELECT COALESCE(SUM(amount), 0)::NUMERIC(12, 2)
      FROM public.ledger_entries WHERE user_type = 'platform'
    ),
    'orders_today', (
      SELECT COUNT(*) FROM public.orders
      WHERE created_at >= date_trunc('day', now())
    ),
    'drivers_payable', (
      SELECT COALESCE(SUM(amount), 0)::NUMERIC(12, 2)
      FROM public.ledger_entries
      WHERE user_type = 'driver' AND amount > 0
    ),
    'restaurants_payable', (
      SELECT COALESCE(SUM(amount), 0)::NUMERIC(12, 2)
      FROM public.ledger_entries
      WHERE user_type = 'restaurant' AND amount > 0
    ),
    'generated_at', now()
  ) INTO v_result;
  RETURN v_result;
END;
$$;

REVOKE ALL ON FUNCTION public.admin_dashboard_metrics() FROM public, anon;
GRANT EXECUTE ON FUNCTION public.admin_dashboard_metrics() TO authenticated;

COMMIT;
```

### Por que manter o fallback `user_metadata.bora_role` em §2.1/§2.2?

Durante a janela de transição, sessões antigas (JWT emitido antes desta migration) ainda têm `bora_role='admin'` no token. O fallback garante que essas sessões continuam a funcionar sem signout forçado. **Removido na fase final** (§7 passo 8 — opcional, com OK adicional).

## §3. Helper Dart `AuthService.isAdmin()`

### Localização
Novo ficheiro: `bora_app/lib/services/auth_admin_service.dart` (separado de `auth_service.dart` existente para isolar a lógica admin e tornar trivial fazer rollback).

### Conteúdo proposto

```dart
import 'package:flutter/foundation.dart';
import 'package:supabase_flutter/supabase_flutter.dart';

/// Single source of truth for "is the current user an admin?".
///
/// Lookup order (3-tier, fail-soft):
///   1. `app_metadata.role == 'admin'`     ← canonical, immutable by client
///   2. `user_metadata.bora_role == 'admin'` ← Phase-1 legacy fallback,
///      kept for one version so JWTs minted before the migration still
///      work without forcing signout.
///   3. Email allowlist                    ← deprecated, debugPrint warning,
///      removed in a follow-up version once telemetry confirms no path
///      relies on it.
///
/// If 1 and 2 return null (claim missing — happens with very old refresh
/// tokens), the caller can opt-in to a one-shot `auth.refreshSession()`
/// before falling back to email.
class AuthAdminService {
  const AuthAdminService._();

  /// Hardcoded fallback emails. Removed in a future version (tracked).
  static const _kDeprecatedEmailAllowlist = <String>{
    'nilofulfarotuga@gmail.com',
    'nilofulfaro@gmail.com',
  };

  /// Returns true if the current Supabase session belongs to an admin.
  ///
  /// Synchronous — reads from the cached session in memory. If the session
  /// is stale, call [refreshAndCheckIsAdmin] instead.
  static bool isAdmin() {
    final user = Supabase.instance.client.auth.currentUser;
    if (user == null) return false;

    // Tier 1: app_metadata.role
    final appRole = user.appMetadata['role'];
    if (appRole is String && appRole == 'admin') return true;

    // Tier 2: user_metadata.bora_role (legacy fallback)
    final userMetadata = user.userMetadata ?? <String, dynamic>{};
    final boraRole = userMetadata['bora_role'];
    if (boraRole is String && boraRole == 'admin') return true;

    // Tier 3: deprecated email allowlist
    final email = user.email;
    if (email != null && _kDeprecatedEmailAllowlist.contains(email)) {
      debugPrint(
        '[AuthAdminService] DEPRECATED: admin check fell back to email '
        'allowlist for "$email". JWT lacked app_metadata.role and '
        'user_metadata.bora_role. Consider signout+signin to refresh JWT.',
      );
      return true;
    }

    return false;
  }

  /// Forces a JWT refresh, then checks isAdmin again. Use this when the
  /// claim is missing AND we suspect the session is stale (e.g. old
  /// refresh token issued before the migration).
  ///
  /// Returns true if either pre-refresh or post-refresh check succeeds.
  static Future<bool> refreshAndCheckIsAdmin() async {
    if (isAdmin()) return true;
    try {
      await Supabase.instance.client.auth.refreshSession();
    } catch (e) {
      debugPrint('[AuthAdminService] refreshSession failed: $e');
    }
    return isAdmin();
  }
}
```

**Notas de design:**
- `static const _kDeprecatedEmailAllowlist` — `Set<String>` para lookup O(1).
- Tipos verificados com `is String` (defense-in-depth contra `userMetadata` malformado).
- Sem dependência de `auth_store.dart` (que faz coisas pesadas) — leitura directa do Supabase client.
- `debugPrint` (não `print`) → só aparece em debug builds.
- Síncrono por default; refresh assíncrono opt-in.
